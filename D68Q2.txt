class Solution {
public:
    string findOrder(vector<string> &dict) {
        int n = dict.size();
        unordered_map<char, vector<char>> adj;
        unordered_map<char, int> indegree;
        
        for (const string& s : dict) {
            for (char c : s) {
                if (indegree.find(c) == indegree.end()) {
                    indegree[c] = 0;
                }
            }
        }

        for (int i = 0; i < n - 1; i++) {
            string s1 = dict[i], s2 = dict[i + 1];
            int len = min(s1.length(), s2.length());
            bool found = false;

            for (int j = 0; j < len; j++) {
                if (s1[j] != s2[j]) {
                    adj[s1[j]].push_back(s2[j]);
                    indegree[s2[j]]++;
                    found = true;
                    break;
                }
            }
            if (!found && s1.length() > s2.length()) return "";
        }

        queue<char> q;
        for (auto const& [ch, count] : indegree) {
            if (count == 0) q.push(ch);
        }

        string result = "";
        while (!q.empty()) {
            char curr = q.front();
            q.pop();
            result += curr;

            for (char neighbor : adj[curr]) {
                indegree[neighbor]--;
                if (indegree[neighbor] == 0) {
                    q.push(neighbor);
                }
            }
        }
        return (result.length() == indegree.size()) ? result : "";
    }
};
